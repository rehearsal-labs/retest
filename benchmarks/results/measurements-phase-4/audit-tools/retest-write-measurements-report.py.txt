def fmt(v, divisor=1, places=2):
    v=[x/divisor for x in v]
    return f'{statistics.median(v):.{places}f} [{min(v):.{places}f}, {max(v):.{places}f}]'
def met(x,key,d=1): return fmt([r[key] for r in x['runs']],d)
def cpu(x,key='runner'):
    return fmt([r[key]['userCpuMs']+r[key]['systemCpuMs'] for r in x['runs']])
def memory(x,key='runner',field='peakRssBytes'):
    return fmt([r[key][field] for r in x['runs']],2**20)
def childcpu(x,key):
    return fmt([sum(t['userCpuMs']+t['systemCpuMs'] for t in r[key]) for r in x['runs']])
def childmem(x,key,field):
    return fmt([max([0]+[t[field] for t in r[key]]) for r in x['runs']],2**20)
def table(headers, rows):
    return '\n'.join(['| '+' | '.join(headers)+' |','| '+' | '.join(['---']*len(headers))+' |']+['| '+' | '.join(map(str,r))+' |' for r in rows])
