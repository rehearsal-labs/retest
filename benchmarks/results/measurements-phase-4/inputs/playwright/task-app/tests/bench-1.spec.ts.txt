import { expect, test } from '@playwright/test'

test('saves task 1', async ({ page }) => {
  await page.goto('/')
  await page.getByTestId('task-title').fill('Release checklist 1')
  await page.getByTestId('save-task').click()
  await expect(page.getByTestId('saved-task')).toHaveText('Release checklist 1')
})
