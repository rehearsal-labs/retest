pidset=set()

eventpidset=set()
def pids(value):
    if isinstance(value,dict):
        for k,v in value.items():
            if k=='pid' and isinstance(v,int): eventpidset.add(v)
            else: pids(v)
    elif isinstance(value,list):
        for v in value: pids(v)
for r in runs:
    native=json.loads(Path(r['resourcesFile']).read_text())
    for p in native['processes']:
        pidset.add(p['pid'])
        paths.add(p['path'])
    events=Path(r['resourcesFile']).parent/'run/events.jsonl'
    if events.exists():
        for line in events.read_text().splitlines():
            e=json.loads(line)
            if e['type']=='browser.started':
                pids(e)
                engine=e.get('engine','unknown')
                builds.setdefault(str(engine),set()).add(json.dumps({k:e.get(k) for k in ['product','version','build']},sort_keys=True))
            if e['type']=='media.started':
                pids(e)
                m=e['media']
                media_builds.add(json.dumps({k:m.get(k) for k in ['version','protocol','build','ffmpeg','encoder']},sort_keys=True))
pidset.update(eventpidset)

def filehash(path):
    p=Path(path)
    h=hashlib.sha256()
    with p.open('rb') as f:
        for chunk in iter(lambda:f.read(1024*1024),b''): h.update(chunk)
    return {'path':str(p),'resolvedPath':str(p.resolve()),'bytes':p.stat().st_size,'sha256':h.hexdigest()}
def treehash(path):
    path=Path(path)
    files=[]
    h=hashlib.sha256()
    for p in sorted(path.rglob('*')):
        if p.is_file():
            item=filehash(p)
            relative=str(p.relative_to(path))
            h.update(relative.encode()+b'\0'+item['sha256'].encode()+b'\0')
            files.append({'relativePath':relative,'bytes':item['bytes'],'sha256':item['sha256']})
    return {'path':str(path),'algorithm':'sha256 of sorted relative path + NUL + file sha256 + NUL','treeSha256':h.hexdigest(),'files':files}
